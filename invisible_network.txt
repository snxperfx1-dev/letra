// ── Cross-TF guard. Pine cannot cheaply pull a *lower* timeframe's full history
//    onto a *higher* chart (it breaks/zeroes detection on Daily+), so any TF
//    below the chart is clamped up to the chart TF and then skipped as a
//    duplicate (see _ok* flags below). Net effect: the network always populates
//    from the chart TF upward — on M1 you get M1→W (the full network); on H4 you
//    get H4→W; on Daily you get D→W. Lower charts = richer network. ──
f_safeTF(simple string _tf) => timeframe.in_seconds(_tf) < timeframe.in_seconds() ? timeframe.period : _tf

// ── FU pool detector (runs per requested timeframe) ──────────────────────────
// FU = a spike through local structure (prior `_lb`-bar high/low) that closes
// back inside ("the middle finger"). The node EXISTS on formation (so detection
// is reliable); a later close beyond the candle body "confirms" it and adds to
// authority. Returns tip, midpoint, direction, present-flag and authority score.
f_fuPool(simple float _wf, simple int _lb) =>
    var float _tip  = na
    var float _bH   = na
    var float _bL   = na
    var float _mid  = na
    var int   _dir  = 0
    var bool  _have = false
    var bool  _conf = false
    float _rng = math.max(high - low, 1e-10)
    float _pHi = ta.highest(high, _lb)[1]
    float _pLo = ta.lowest(low,  _lb)[1]
    bool  _bear = not na(_pHi) and high > _pHi and close < _pHi and (high - math.max(open, close)) / _rng >= _wf
    bool  _bull = not na(_pLo) and low  < _pLo and close > _pLo and (math.min(open, close) - low) / _rng >= _wf
    if _bear
        _dir  := -1
        _tip  := high
        _bH   := math.max(open, close)
        _bL   := math.min(open, close)
        _mid  := _bH + (_tip - _bH) * 0.5
        _have := true
        _conf := false
    else if _bull
        _dir  := 1
        _tip  := low
        _bH   := math.max(open, close)
        _bL   := math.min(open, close)
        _mid  := _tip + (_bL - _tip) * 0.5
        _have := true
        _conf := false
    if _have and _dir == -1 and not _conf and close < _bL
        _conf := true
    if _have and _dir ==  1 and not _conf and close > _bH
        _conf := true
    float _wk = (_dir == -1 and _have) ? (_tip - _bH) / math.max(ta.atr(14), 1e-10) : (_dir == 1 and _have) ? (_bL - _tip) / math.max(ta.atr(14), 1e-10) : 0.0
    float _score = 20.0 + math.min(25.0, _wk * 15.0) + (_conf ? 30.0 : 0.0) + (_wk > 1.0 ? 15.0 : 0.0) + (_wk > 1.5 ? 10.0 : 0.0)
    [_have ? _tip : na, _mid, _dir, _have ? 1 : 0, _score]

// ── Lower-TF picker. request.security_lower_tf returns one value per intrabar
//    of the current chart bar (empty array when the requested TF >= chart). We
//    take the most recent valid FU among those intrabars so a sub-chart FU
//    (e.g. M5 while on H4) still becomes a node — Engine 2.0 global visibility. ──
f_ltfPick(array<float> _t, array<float> _m, array<int> _d, array<int> _v, array<float> _s) =>
    float _tip = na
    float _mid = na
    float _sc  = 0.0
    int   _dir = 0
    int   _val = 0
    if array.size(_t) > 0
        for _j = array.size(_t) - 1 to 0
            if _val == 0 and array.get(_v, _j) == 1 and not na(array.get(_t, _j))
                _tip := array.get(_t, _j)
                _mid := array.get(_m, _j)
                _dir := array.get(_d, _j)
                _sc  := array.get(_s, _j)
                _val := 1
    [_tip, _mid, _dir, _sc, _val]

// Effective authority = sub-score + timeframe weight*4 + revisits*3.
f_auth(int _i) => array.get(nSc, _i) + array.get(nWt, _i) * 4.0 + array.get(nRev, _i) * 3.0

// Magnet score = authority + proximity + bias-alignment + untouched/recursion.
f_magnet(int _i) =>
    float _d  = math.abs(close - array.get(nPx, _i)) / math.max(_atr, 1e-10)
    float _db = math.max(0.0, 30.0 - _d * 3.0)
    float _ab = (array.get(nDir, _i) == netBias and netBias != 0) ? 15.0 : 0.0
    float _ub = array.get(nRev, _i) == 0 ? 10.0 : math.min(20.0, array.get(nRev, _i) * 3.0)
    f_auth(_i) * 0.5 + _db + _ab + _ub

// Collect eligible open nodes on one side of price, sorted nearest-first.
//   _aheadSide = true  → nodes in the bias direction (where price is GOING)
//   _aheadSide = false → nodes opposite the bias (the trail price CAME FROM)
f_pathNodes(bool _aheadSide) =>
    array<int> _r = array.new_int()
    if array.size(nPx) > 0
        for _i = 0 to array.size(nPx) - 1
            float _np    = array.get(nPx, _i)
            bool  _ahead = netBias == 1 ? _np > close : _np < close
            if array.get(nState, _i) != 2 and f_auth(_i) >= authMin and (_aheadSide ? _ahead : not _ahead)
                array.push(_r, _i)
    if array.size(_r) > 1
        for _a = 1 to array.size(_r) - 1
            int   _key = array.get(_r, _a)
            float _kd  = math.abs(close - array.get(nPx, _key))
            int   _b   = _a - 1
            bool  _go  = _b >= 0
            while _go
                if math.abs(close - array.get(nPx, array.get(_r, _b))) > _kd
                    array.set(_r, _b + 1, array.get(_r, _b))
                    _b -= 1
                    _go := _b >= 0
                else
                    _go := false
            array.set(_r, _b + 1, _key)
    _r

    int   _domIdx        = -1
    float _domAuth       = 0.0
    float _fezHiPx       = na   // strongest node above price (upper flip)
    float _fezLoPx       = na   // strongest node below price (lower flip)
    int   _fezHiW        = 0
    int   _fezLoW        = 0
    float _fezHiAuth     = 0.0
    float _fezLoAuth     = 0.0
    if array.size(nPx) > 0
        for _i = 0 to array.size(nPx) - 1
            int   _st = array.get(nState, _i)
            if _st == 2
                _consumedCnt += 1
            else
                _openMem += 1
            float _a  = f_auth(_i)
            if _st != 2 and _a >= authMin
                float _np = array.get(nPx, _i)
                int   _nd = array.get(nDir, _i)
                int   _wt = array.get(nWt, _i)
                if showLevels
                    color _base = tfColors ? f_tfColor(_wt) : (_nd == 1 ? #00ffaa : #ff6600)
                    color _c    = _st == 0 ? color.new(_base, 25) : color.new(_base, 70)
                    array.push(level_lines, line.new(bar_index - 1, _np, bar_index + 12, _np, color = _c, width = 1, style = _st == 0 ? line.style_solid : line.style_dotted))
                    array.push(level_lbls, label.new(bar_index + 12, _np, f_wtTf(_wt) + (_nd == 1 ? " ▲ " : " ▼ ") + str.tostring(math.round(_a)), style = label.style_label_left, color = color.new(#10103a, 20), textcolor = _st == 0 ? _base : color.gray, size = size.tiny))
                _activeTotal += 1
                if array.get(nWt, _i) >= 7
                    _objCnt += 1
                if _nd == netBias and netBias != 0
                    _activeAligned += 1
                if _nd == 1
                    _bullAuth += _a
                else if _nd == -1
                    _bearAuth += _a
                if _a > _topAuth
                    _topAuth := _a
                if _a > _domAuth
                    _domAuth := _a
                    _domIdx  := _i
                // FEZ corridor = between the strongest LINKED structures, not
                // merely the nearest: top-authority node above & below price.
                if _np > close and _a > _fezHiAuth
                    _fezHiPx   := _np
                    _fezHiW    := _wt
                    _fezHiAuth := _a
                if _np < close and _a > _fezLoAuth
                    _fezLoPx   := _np
                    _fezLoW    := _wt
                    _fezLoAuth := _a

    // ── FEZ (FU Equilibrium Zone) — the corridor price oscillates inside,
    //    bounded by the strongest linked structure above and below price. ──────
    if showFEZ and not na(_fezHiPx) and not na(_fezLoPx)
        fez_box := box.new(bar_index - 1, _fezHiPx, bar_index + 12, _fezLoPx, border_color = color.new(#7b2fff, 40), border_width = 1, bgcolor = color.new(#7b2fff, 88))
        float _comp = (_fezHiPx - _fezLoPx) / math.max(_atr, 1e-10)
        fez_lbl := label.new(bar_index - 1, (_fezHiPx + _fezLoPx) * 0.5, "FEZ  " + f_wtTf(_fezLoW) + "↔" + f_wtTf(_fezHiW) + "  " + str.tostring(_comp, "#.#") + " ATR", style = label.style_label_right, color = color.new(#3a1f6b, 20), textcolor = #d9c6ff, size = size.tiny)

    // ── Routes: CURRENT path (forward, bias side) + PREVIOUS path (trail) ────
    float _routeConf = (netBias == 0 or _activeTotal == 0) ? 0.0 : math.min(100.0, _topAuth * 0.5 + (_activeAligned / math.max(_activeTotal, 1) * 100.0) * 0.5)
    string _domPath = "—"
    float _altConfOut = 0.0

    // CURRENT path — where price is going: chain forward (to the right) through
    // bias-side open nodes, nearest-first. Color = target TF, width = authority.
    if showCurPath and netBias != 0 and _routeConf >= routeMinConf
        array<int> _ri = f_pathNodes(true)
        if array.size(_ri) > 0
            int   _x = bar_index
            float _y = close
            _domPath := "Price"
            for _i = 0 to array.size(_ri) - 1
                int   _idx = array.get(_ri, _i)
                float _n   = array.get(nPx, _idx)
                int   _wt  = array.get(nWt, _idx)
                float _aa  = f_auth(_idx)
                int   _x2  = _x + 9
                int   _w   = _aa >= 90 ? 4 : _aa >= 65 ? 3 : 2
                color _lc  = tfColors ? f_tfColor(_wt) : #b388ff
                array.push(route_lines, line.new(_x, _y, _x2, _n, color = color.new(_lc, 0), width = _w, style = line.style_arrow_right))
                _x := _x2
                _y := _n
                _domPath := _domPath + " → " + f_wtTf(_wt)
            route_lbl := label.new(_x, _y, "CURRENT  " + str.tostring(math.round(_routeConf)) + "%  →  " + str.tostring(math.round_to_mintick(_y)), style = label.style_label_left, color = color.new(#7b2fff, 5), textcolor = color.white, size = size.normal)

    // ALTERNATIVE / AFE path — the reversal scenario: a dashed leg to the
    // strongest node OPPOSITE the bias side, with the complementary probability.
    if showAltPath and netBias != 0 and array.size(nPx) > 0
        int   _altIdx = -1
        float _altA   = 0.0
        for _i = 0 to array.size(nPx) - 1
            if array.get(nState, _i) != 2 and f_auth(_i) >= authMin
                float _np  = array.get(nPx, _i)
                bool  _opp = netBias == 1 ? _np < close : _np > close
                if _opp and f_auth(_i) > _altA
                    _altA   := f_auth(_i)
                    _altIdx := _i
        if _altIdx >= 0
            float _an     = array.get(nPx, _altIdx)
            float _altConf = math.max(0.0, 100.0 - _routeConf)
            _altConfOut := _altConf
            array.push(route_lines, line.new(bar_index, close, bar_index + 9, _an, color = color.new(#ff8a3d, 10), width = 1, style = line.style_dashed))
            alt_lbl := label.new(bar_index + 9, _an, "ALT  " + str.tostring(math.round(_altConf)) + "%  →  " + str.tostring(math.round_to_mintick(_an)), style = label.style_label_left, color = color.new(#6b3a14, 10), textcolor = #ffb27f, size = size.small)

    // PREVIOUS path — the trail price came from: chain backward (to the left)
    // through the open nodes behind price, nearest-first, as dim cyan arrows.
    if showPrevPath and netBias != 0
        array<int> _pi = f_pathNodes(false)
        if array.size(_pi) > 0
            int   _x = bar_index
            float _y = close
            for _i = 0 to array.size(_pi) - 1
                int   _idx = array.get(_pi, _i)
                float _n   = array.get(nPx, _idx)
                int   _x2  = _x - 9
                array.push(route_lines, line.new(_x2, _n, _x, _y, color = color.new(#26c6da, 15), width = 2, style = line.style_arrow_right))
                _x := _x2
                _y := _n
            prev_lbl := label.new(_x, _y, "TRAIL  ←  " + str.tostring(math.round_to_mintick(_y)), style = label.style_label_right, color = color.new(#0d5b66, 5), textcolor = #aef3fb, size = size.small)

    // ── Network panel (Engine 2.13) ─────────────────────────────────────────
    if showPanel
        if na(netPanel)
            netPanel := table.new(position.top_right, 2, 22, bgcolor = color.new(#0b0b1c, 6), frame_color = color.new(#7b2fff, 30), frame_width = 2, border_width = 0)
        int   _nodesTot = array.size(nPx)
        int   _fwd      = array.size(f_pathNodes(true))
        int   _bwd      = array.size(f_pathNodes(false))
        int   _convCnt  = math.max(0, _activeTotal - 1)
        int   _pathsAct = _fwd + _bwd + (showAltPath ? 1 : 0)
        float _alSum    = _bullAuth + _bearAuth
        int   _bullPct  = _alSum > 0 ? math.round(_bullAuth / _alSum * 100.0) : 0
        int   _confPct  = 100 - _bullPct
        string _domTf   = _domIdx >= 0 ? f_wtTf(array.get(nWt, _domIdx)) : "—"
        string _domTgt  = _domIdx >= 0 ? str.tostring(math.round_to_mintick(array.get(nPx, _domIdx))) : "—"
        table.cell(netPanel, 0, 0, "INVISIBLE NETWORK", text_color = #b388ff, text_size = size.normal)
        table.cell(netPanel, 1, 0, netBias == 1 ? "▲ BULL" : netBias == -1 ? "▼ BEAR" : "○ FLAT", text_color = netBias == 1 ? #5aff9e : netBias == -1 ? #ff5a5a : color.gray, text_size = size.normal)
        table.cell(netPanel, 0, 1, "Nodes / Conversations", text_color = color.silver, text_size = size.small)
        table.cell(netPanel, 1, 1, str.tostring(_nodesTot) + " · " + str.tostring(_convCnt), text_color = color.white, text_size = size.small)
        table.cell(netPanel, 0, 2, "Active paths", text_color = color.silver, text_size = size.small)
        table.cell(netPanel, 1, 2, str.tostring(_pathsAct), text_color = color.white, text_size = size.small)
        table.cell(netPanel, 0, 3, "Dominant target / TF", text_color = color.silver, text_size = size.small)
        table.cell(netPanel, 1, 3, _domTgt + "  " + _domTf, text_color = #ffd000, text_size = size.small)
        table.cell(netPanel, 0, 4, "Alignment / Conflict", text_color = color.silver, text_size = size.small)
        table.cell(netPanel, 1, 4, "Bull " + str.tostring(_bullPct) + "% · " + str.tostring(_confPct) + "%", text_color = _bullPct >= 50 ? #5aff9e : #ff5a5a, text_size = size.small)
        table.cell(netPanel, 0, 5, "Path confidence", text_color = color.silver, text_size = size.small)
        table.cell(netPanel, 1, 5, str.tostring(math.round(_routeConf)) + "%", text_color = _routeConf >= 60 ? #5aff9e : color.yellow, text_size = size.small)
        table.cell(netPanel, 0, 6, "Memory", text_color = color.silver, text_size = size.small)
        table.cell(netPanel, 1, 6, str.tostring(_openMem) + " open · " + str.tostring(_consumedCnt) + " consumed", text_color = color.white, text_size = size.small)
        // magnet-sorted list of open eligible nodes (used for both sections)
        array<int> _mg = array.new_int()
        if array.size(nPx) > 0
            for _i = 0 to array.size(nPx) - 1
                if array.get(nState, _i) != 2 and f_auth(_i) >= authMin
                    array.push(_mg, _i)
        if array.size(_mg) > 1
            for _a = 1 to array.size(_mg) - 1
                int   _key = array.get(_mg, _a)
                float _km  = f_magnet(_key)
                int   _b   = _a - 1
                bool  _go  = _b >= 0
                while _go
                    if f_magnet(array.get(_mg, _b)) < _km
                        array.set(_mg, _b + 1, array.get(_mg, _b))
                        _b -= 1
                        _go := _b >= 0
                    else
                        _go := false
                array.set(_mg, _b + 1, _key)
        // STRONGEST MAGNETS (top 3 by magnet score)
        table.cell(netPanel, 0, 7, "─ STRONGEST MAGNETS ─", text_color = #ff7fe6, text_size = size.small)
        table.cell(netPanel, 1, 7, "", text_color = color.white, text_size = size.small)
        int _mn = math.min(array.size(_mg), 3)
        for _k = 0 to 2
            if _k < _mn
                int _idx = array.get(_mg, _k)
                int _nd  = array.get(nDir, _idx)
                table.cell(netPanel, 0, 8 + _k, str.tostring(_k + 1) + " " + f_wtTf(array.get(nWt, _idx)) + (_nd == 1 ? " ▲ " : " ▼ ") + str.tostring(math.round_to_mintick(array.get(nPx, _idx))), text_color = tfColors ? f_tfColor(array.get(nWt, _idx)) : (_nd == 1 ? #5aff9e : #ff5a5a), text_size = size.small)
                table.cell(netPanel, 1, 8 + _k, "MAGNET " + str.tostring(math.round(f_magnet(_idx))), text_color = #ffd000, text_size = size.small)
            else
                table.cell(netPanel, 0, 8 + _k, "—", text_color = color.gray, text_size = size.small)
                table.cell(netPanel, 1, 8 + _k, "", text_color = color.gray, text_size = size.small)
        // ACTIVE CONVERSATIONS (top 5, with status + distance)
        table.cell(netPanel, 0, 11, "─ ACTIVE CONVERSATIONS ─", text_color = #ff7fe6, text_size = size.small)
        table.cell(netPanel, 1, 11, "", text_color = color.white, text_size = size.small)
        int _cn = math.min(array.size(_mg), 5)
        for _k = 0 to 4
            if _k < _cn
                int   _idx  = array.get(_mg, _k)
                int   _nd   = array.get(nDir, _idx)
                float _np   = array.get(nPx, _idx)
                float _dATR = math.abs(close - _np) / math.max(_atr, 1e-10)
                int   _rv   = array.get(nRev, _idx)
                bool  _ahead = netBias == 1 ? _np > close : _np < close
                string _status = _dATR < 0.5 ? "ATTACKING" : _rv > 0 ? "RETURNING" : _ahead ? "ATTRACTING" : "BEHIND"
                table.cell(netPanel, 0, 12 + _k, f_wtTf(array.get(nWt, _idx)) + (_nd == 1 ? " ▲ " : " ▼ ") + str.tostring(math.round_to_mintick(_np)), text_color = tfColors ? f_tfColor(array.get(nWt, _idx)) : (_nd == 1 ? #5aff9e : #ff5a5a), text_size = size.small)
                table.cell(netPanel, 1, 12 + _k, _status + "  " + str.tostring(_dATR, "#.#") + " ATR", text_color = color.white, text_size = size.small)
            else
                table.cell(netPanel, 0, 12 + _k, "—", text_color = color.gray, text_size = size.small)
                table.cell(netPanel, 1, 12 + _k, "", text_color = color.gray, text_size = size.small)
        // DOMINANT PATH
        table.cell(netPanel, 0, 17, "─ DOMINANT PATH ─", text_color = #ff7fe6, text_size = size.small)
        table.cell(netPanel, 1, 17, str.tostring(math.round(_routeConf)) + "%", text_color = _routeConf >= 60 ? #5aff9e : color.yellow, text_size = size.small)
        table.cell(netPanel, 0, 18, _domPath, text_color = #b388ff, text_size = size.small)
        table.cell(netPanel, 1, 18, "", text_color = color.white, text_size = size.small)
        int _fezCnt = (not na(_fezHiPx) and not na(_fezLoPx)) ? 1 : 0
        table.cell(netPanel, 0, 19, "Active FU / FEZ", text_color = color.silver, text_size = size.small)
        table.cell(netPanel, 1, 19, str.tostring(_activeTotal) + " · " + str.tostring(_fezCnt), text_color = color.white, text_size = size.small)
        table.cell(netPanel, 0, 20, "Objectives / Edges", text_color = color.silver, text_size = size.small)
        table.cell(netPanel, 1, 20, str.tostring(_objCnt) + " · " + str.tostring(_convCnt), text_color = color.white, text_size = size.small)
        table.cell(netPanel, 0, 21, "Alt path probability", text_color = color.silver, text_size = size.small)
        table.cell(netPanel, 1, 21, str.tostring(math.round(_altConfOut)) + "%", text_color = #ffb27f, text_size = size.small)
