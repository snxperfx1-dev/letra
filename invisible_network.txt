// INVISIBLE NETWORK  (Engine 0 / 1A.13)  —  standalone FU conversation graph
// ───────────────────────────────────────────────────────────────────────────────
// Price is not moving between swings or BOS — it moves between remembered
// manipulations (validated FU candles). This indicator scans every timeframe
// (M1 → Weekly), remembers each validated FU as a persistent network NODE
// (ACTIVE → DORMANT → CONSUMED → HISTORICAL, with revisit counting), ranks them by
// authority, and reconstructs the live CONVERSATION between them — drawing the
// strongest active route as connected arrows plus the remembered network levels,
// and listing the active conversations in a panel.
// ═══════════════════════════════════════════════════════════════════════════════
indicator("Invisible Network", overlay = true, max_lines_count = 500, max_labels_count = 500, max_boxes_count = 500)

// ── Inputs ───────────────────────────────────────────────────────────────────
string GRP = "Invisible Network"
float  wickFrac     = input.float(0.4, "FU spike: min wick / range",        group = GRP, minval = 0.2, maxval = 0.9, step = 0.05)
int    lookback     = input.int  (3,   "FU spike: structure lookback",      group = GRP, minval = 2, maxval = 30)
int    authMin      = input.int  (45,  "Min node authority to show",        group = GRP, minval = 0, maxval = 120)
int    routeMinConf = input.int  (35,  "Min path confidence % (route)",     group = GRP, minval = 0, maxval = 100)
int    nodeMax      = input.int  (64,  "Max remembered nodes",              group = GRP, minval = 10, maxval = 200)
bool   showLevels   = input.bool (true,  "Show network levels",             group = GRP)
bool   showCurPath  = input.bool (true,  "Show CURRENT path (forward route)",group = GRP)
bool   showPrevPath = input.bool (true,  "Show PREVIOUS path (trail behind)",group = GRP)
bool   showPanel    = input.bool (true,  "Show conversation panel",         group = GRP)
int    dormantBars  = input.int  (120, "Bars until DORMANT",                group = GRP, minval = 20, maxval = 2000)
int    historyBars  = input.int  (600, "Bars until HISTORICAL",             group = GRP, minval = 100, maxval = 5000)

// ── Always scan the real timeframe. The whole network must stay visible no
//    matter which chart timeframe the user is on, so we NEVER clamp a lower TF
//    up to the chart TF — every magnet (M1 → W) is detected and remembered on
//    every chart. (Pass-through kept as a function so call sites are stable.) ──
f_safeTF(simple string _tf) => _tf

// ── FU pool detector (runs per requested timeframe) ──────────────────────────
// FU = a spike through local structure (prior `_lb`-bar high/low) that closes
// back inside ("the middle finger"). The node EXISTS on formation (so detection
// is reliable); a later close beyond the candle body "confirms" it and adds to
// authority. Returns tip, midpoint, direction, present-flag and authority score.
f_fuPool(simple float _wf, simple int _lb) =>
    var float _tip  = na
    var float _bH   = na
    var float _bL   = na
    var float _mid  = na
    var int   _dir  = 0
    var bool  _have = false
    var bool  _conf = false
    float _rng = math.max(high - low, 1e-10)
    float _pHi = ta.highest(high, _lb)[1]
    float _pLo = ta.lowest(low,  _lb)[1]
    bool  _bear = not na(_pHi) and high > _pHi and close < _pHi and (high - math.max(open, close)) / _rng >= _wf
    bool  _bull = not na(_pLo) and low  < _pLo and close > _pLo and (math.min(open, close) - low) / _rng >= _wf
    if _bear
        _dir  := -1
        _tip  := high
        _bH   := math.max(open, close)
        _bL   := math.min(open, close)
        _mid  := _bH + (_tip - _bH) * 0.5
        _have := true
        _conf := false
    else if _bull
        _dir  := 1
        _tip  := low
        _bH   := math.max(open, close)
        _bL   := math.min(open, close)
        _mid  := _tip + (_bL - _tip) * 0.5
        _have := true
        _conf := false
    if _have and _dir == -1 and not _conf and close < _bL
        _conf := true
    if _have and _dir ==  1 and not _conf and close > _bH
        _conf := true
    float _wk = (_dir == -1 and _have) ? (_tip - _bH) / math.max(ta.atr(14), 1e-10) : (_dir == 1 and _have) ? (_bL - _tip) / math.max(ta.atr(14), 1e-10) : 0.0
    float _score = 20.0 + math.min(25.0, _wk * 15.0) + (_conf ? 30.0 : 0.0) + (_wk > 1.0 ? 15.0 : 0.0) + (_wk > 1.5 ? 10.0 : 0.0)
    [_have ? _tip : na, _mid, _dir, _have ? 1 : 0, _score]

// ── Scan all eight timeframes ────────────────────────────────────────────────
[wTip,   wMid,   wDir,   wV,   wSc]   = request.security(syminfo.tickerid, f_safeTF("W"),   f_fuPool(wickFrac, lookback), lookahead = barmerge.lookahead_off)
[dTip,   dMid,   dDir,   dV,   dSc]   = request.security(syminfo.tickerid, f_safeTF("D"),   f_fuPool(wickFrac, lookback), lookahead = barmerge.lookahead_off)
[h4Tip,  h4Mid,  h4Dir,  h4V,  h4Sc]  = request.security(syminfo.tickerid, f_safeTF("240"), f_fuPool(wickFrac, lookback), lookahead = barmerge.lookahead_off)
[h1Tip,  h1Mid,  h1Dir,  h1V,  h1Sc]  = request.security(syminfo.tickerid, f_safeTF("60"),  f_fuPool(wickFrac, lookback), lookahead = barmerge.lookahead_off)
[m15Tip, m15Mid, m15Dir, m15V, m15Sc] = request.security(syminfo.tickerid, f_safeTF("15"),  f_fuPool(wickFrac, lookback), lookahead = barmerge.lookahead_off)
[m5Tip,  m5Mid,  m5Dir,  m5V,  m5Sc]  = request.security(syminfo.tickerid, f_safeTF("5"),   f_fuPool(wickFrac, lookback), lookahead = barmerge.lookahead_off)
[m3Tip,  m3Mid,  m3Dir,  m3V,  m3Sc]  = request.security(syminfo.tickerid, f_safeTF("3"),   f_fuPool(wickFrac, lookback), lookahead = barmerge.lookahead_off)
[m1Tip,  m1Mid,  m1Dir,  m1V,  m1Sc]  = request.security(syminfo.tickerid, f_safeTF("1"),   f_fuPool(wickFrac, lookback), lookahead = barmerge.lookahead_off)

// ── Bias: the highest-timeframe live validated FU sets the directional context;
//    fall back to a 50-EMA slope when no FU is active. ─────────────────────────
float _ema = ta.ema(close, 50)
int   _trendBias = close > _ema ? 1 : close < _ema ? -1 : 0
int   netBias =
     nz(wV)   == 1 ? nz(wDir)   :
     nz(dV)   == 1 ? nz(dDir)   :
     nz(h4V)  == 1 ? nz(h4Dir)  :
     nz(h1V)  == 1 ? nz(h1Dir)  :
     nz(m15V) == 1 ? nz(m15Dir) :
     nz(m5V)  == 1 ? nz(m5Dir)  : _trendBias

// ── Persistent network registry — nodes are never deleted, only transitioned.
//    state: 0 ACTIVE · 1 DORMANT · 2 CONSUMED · 3 HISTORICAL ──────────────────
var array<float> nPx    = array.new_float()
var array<float> nMid   = array.new_float()
var array<int>   nDir   = array.new_int()
var array<float> nSc    = array.new_float()
var array<int>   nWt    = array.new_int()    // timeframe authority weight 1..8
var array<int>   nState = array.new_int()
var array<int>   nBar   = array.new_int()
var array<int>   nRev   = array.new_int()

f_wtTf(int _wt) => _wt == 8 ? "W" : _wt == 7 ? "D" : _wt == 6 ? "H4" : _wt == 5 ? "H1" : _wt == 4 ? "M15" : _wt == 3 ? "M5" : _wt == 2 ? "M3" : "M1"

f_nodeAdd(float _tip, float _mid, int _dir, float _sc, int _wt) =>
    array.push(nPx, _tip)
    array.push(nMid, _mid)
    array.push(nDir, _dir)
    array.push(nSc, _sc)
    array.push(nWt, _wt)
    array.push(nState, 0)
    array.push(nBar, bar_index)
    array.push(nRev, 0)
    if array.size(nPx) > nodeMax
        array.shift(nPx)
        array.shift(nMid)
        array.shift(nDir)
        array.shift(nSc)
        array.shift(nWt)
        array.shift(nState)
        array.shift(nBar)
        array.shift(nRev)

// Remember a NEW validated FU per timeframe (tip change = a freshly born node).
var float _pvW   = na
var float _pvD   = na
var float _pvH4  = na
var float _pvH1  = na
var float _pvM15 = na
var float _pvM5  = na
var float _pvM3  = na
var float _pvM1  = na
if nz(wV)   == 1 and not na(wTip)   and (na(_pvW)   or wTip   != _pvW)
    f_nodeAdd(wTip,   wMid,   nz(wDir),   nz(wSc),   8)
    _pvW := wTip
if nz(dV)   == 1 and not na(dTip)   and (na(_pvD)   or dTip   != _pvD)
    f_nodeAdd(dTip,   dMid,   nz(dDir),   nz(dSc),   7)
    _pvD := dTip
if nz(h4V)  == 1 and not na(h4Tip)  and (na(_pvH4)  or h4Tip  != _pvH4)
    f_nodeAdd(h4Tip,  h4Mid,  nz(h4Dir),  nz(h4Sc),  6)
    _pvH4 := h4Tip
if nz(h1V)  == 1 and not na(h1Tip)  and (na(_pvH1)  or h1Tip  != _pvH1)
    f_nodeAdd(h1Tip,  h1Mid,  nz(h1Dir),  nz(h1Sc),  5)
    _pvH1 := h1Tip
if nz(m15V) == 1 and not na(m15Tip) and (na(_pvM15) or m15Tip != _pvM15)
    f_nodeAdd(m15Tip, m15Mid, nz(m15Dir), nz(m15Sc), 4)
    _pvM15 := m15Tip
if nz(m5V)  == 1 and not na(m5Tip)  and (na(_pvM5)  or m5Tip  != _pvM5)
    f_nodeAdd(m5Tip,  m5Mid,  nz(m5Dir),  nz(m5Sc),  3)
    _pvM5 := m5Tip
if nz(m3V)  == 1 and not na(m3Tip)  and (na(_pvM3)  or m3Tip  != _pvM3)
    f_nodeAdd(m3Tip,  m3Mid,  nz(m3Dir),  nz(m3Sc),  2)
    _pvM3 := m3Tip
if nz(m1V)  == 1 and not na(m1Tip)  and (na(_pvM1)  or m1Tip  != _pvM1)
    f_nodeAdd(m1Tip,  m1Mid,  nz(m1Dir),  nz(m1Sc),  1)
    _pvM1 := m1Tip

// State transitions + revisit counting (every bar, to maintain memory).
float _atr = ta.atr(14)
if array.size(nPx) > 0
    for _i = 0 to array.size(nPx) - 1
        if array.get(nState, _i) != 2
            float _np  = array.get(nPx, _i)
            int   _nd  = array.get(nDir, _i)
            int   _age = bar_index - array.get(nBar, _i)
            if (_nd == -1 ? high > _np : low < _np)
                array.set(nState, _i, 2)                         // CONSUMED — price traded through it
            else
                if math.abs(close - _np) < _atr * 0.25
                    array.set(nRev, _i, array.get(nRev, _i) + 1) // revisit (defense) raises authority
                array.set(nState, _i, _age > historyBars ? 3 : _age > dormantBars ? 1 : 0)

// Effective authority = sub-score + timeframe weight*4 + revisits*3.
f_auth(int _i) => array.get(nSc, _i) + array.get(nWt, _i) * 4.0 + array.get(nRev, _i) * 3.0

// Magnet score = authority + proximity + bias-alignment + untouched/recursion.
f_magnet(int _i) =>
    float _d  = math.abs(close - array.get(nPx, _i)) / math.max(_atr, 1e-10)
    float _db = math.max(0.0, 30.0 - _d * 3.0)
    float _ab = (array.get(nDir, _i) == netBias and netBias != 0) ? 15.0 : 0.0
    float _ub = array.get(nRev, _i) == 0 ? 10.0 : math.min(20.0, array.get(nRev, _i) * 3.0)
    f_auth(_i) * 0.5 + _db + _ab + _ub

// Collect eligible open nodes on one side of price, sorted nearest-first.
//   _aheadSide = true  → nodes in the bias direction (where price is GOING)
//   _aheadSide = false → nodes opposite the bias (the trail price CAME FROM)
f_pathNodes(bool _aheadSide) =>
    array<int> _r = array.new_int()
    if array.size(nPx) > 0
        for _i = 0 to array.size(nPx) - 1
            float _np    = array.get(nPx, _i)
            bool  _ahead = netBias == 1 ? _np > close : _np < close
            if array.get(nState, _i) <= 1 and f_auth(_i) >= authMin and (_aheadSide ? _ahead : not _ahead)
                array.push(_r, _i)
    if array.size(_r) > 1
        for _a = 1 to array.size(_r) - 1
            int   _key = array.get(_r, _a)
            float _kd  = math.abs(close - array.get(nPx, _key))
            int   _b   = _a - 1
            bool  _go  = _b >= 0
            while _go
                if math.abs(close - array.get(nPx, array.get(_r, _b))) > _kd
                    array.set(_r, _b + 1, array.get(_r, _b))
                    _b -= 1
                    _go := _b >= 0
                else
                    _go := false
            array.set(_r, _b + 1, _key)
    _r

// ── Drawing objects (cleared & rebuilt on the last bar) ──────────────────────
var array<line>  route_lines = array.new<line>()
var array<line>  level_lines = array.new<line>()
var array<label> level_lbls  = array.new<label>()
var label        route_lbl   = na
var label        prev_lbl    = na
var table        netPanel    = na

if barstate.islast
    if array.size(route_lines) > 0
        for _i = 0 to array.size(route_lines) - 1
            line.delete(array.get(route_lines, _i))
        array.clear(route_lines)
    if array.size(level_lines) > 0
        for _i = 0 to array.size(level_lines) - 1
            line.delete(array.get(level_lines, _i))
        array.clear(level_lines)
    if array.size(level_lbls) > 0
        for _i = 0 to array.size(level_lbls) - 1
            label.delete(array.get(level_lbls, _i))
        array.clear(level_lbls)
    if not na(route_lbl)
        label.delete(route_lbl)
    if not na(prev_lbl)
        label.delete(prev_lbl)

    // ── Network levels + tallies (open/consumed counts, route inputs) ────────
    int   _activeAligned = 0
    int   _activeTotal   = 0
    float _topAuth       = 0.0
    int   _openMem       = 0
    int   _consumedCnt   = 0
    if array.size(nPx) > 0
        for _i = 0 to array.size(nPx) - 1
            int   _st = array.get(nState, _i)
            if _st == 2
                _consumedCnt += 1
            else
                _openMem += 1
            float _a  = f_auth(_i)
            if _st <= 1 and _a >= authMin
                float _np = array.get(nPx, _i)
                int   _nd = array.get(nDir, _i)
                if showLevels
                    color _c = _st == 0 ? (_nd == 1 ? color.new(#00ffaa, 25) : color.new(#ff6600, 25)) : color.new(color.gray, 60)
                    array.push(level_lines, line.new(bar_index - 1, _np, bar_index + 12, _np, color = _c, width = 1, style = _st == 0 ? line.style_solid : line.style_dotted))
                    array.push(level_lbls, label.new(bar_index + 12, _np, f_wtTf(array.get(nWt, _i)) + (_nd == 1 ? " ▲ " : " ▼ ") + str.tostring(math.round(_a)), style = label.style_label_left, color = color.new(#10103a, 20), textcolor = _st == 0 ? (_nd == 1 ? #00ffaa : #ff6600) : color.gray, size = size.tiny))
                if _st == 0
                    _activeTotal += 1
                    if _nd == netBias and netBias != 0
                        _activeAligned += 1
                    if _a > _topAuth
                        _topAuth := _a

    // ── Routes: CURRENT path (forward, bias side) + PREVIOUS path (trail) ────
    float _routeConf = (netBias == 0 or _activeTotal == 0) ? 0.0 : math.min(100.0, _topAuth * 0.5 + (_activeAligned / math.max(_activeTotal, 1) * 100.0) * 0.5)
    string _domPath = "—"

    // CURRENT path — where price is going: chain forward (to the right) through
    // bias-side open nodes, nearest-first, as connected purple arrows.
    if showCurPath and netBias != 0 and _routeConf >= routeMinConf
        array<int> _ri = f_pathNodes(true)
        if array.size(_ri) > 0
            int   _x = bar_index
            float _y = close
            _domPath := "Price"
            for _i = 0 to array.size(_ri) - 1
                int   _idx = array.get(_ri, _i)
                float _n   = array.get(nPx, _idx)
                int   _x2  = _x + 9
                array.push(route_lines, line.new(_x, _y, _x2, _n, color = color.new(#b388ff, 0), width = 2, style = line.style_arrow_right))
                _x := _x2
                _y := _n
                _domPath := _domPath + " → " + f_wtTf(array.get(nWt, _idx))
            route_lbl := label.new(_x, _y, "CURRENT  " + str.tostring(math.round(_routeConf)) + "%  →  " + str.tostring(math.round_to_mintick(_y)), style = label.style_label_left, color = color.new(#7b2fff, 5), textcolor = color.white, size = size.normal)

    // PREVIOUS path — the trail price came from: chain backward (to the left)
    // through the open nodes behind price, nearest-first, as dim cyan arrows.
    if showPrevPath and netBias != 0
        array<int> _pi = f_pathNodes(false)
        if array.size(_pi) > 0
            int   _x = bar_index
            float _y = close
            for _i = 0 to array.size(_pi) - 1
                int   _idx = array.get(_pi, _i)
                float _n   = array.get(nPx, _idx)
                int   _x2  = _x - 9
                array.push(route_lines, line.new(_x2, _n, _x, _y, color = color.new(#26c6da, 15), width = 2, style = line.style_arrow_right))
                _x := _x2
                _y := _n
            prev_lbl := label.new(_x, _y, "TRAIL  ←  " + str.tostring(math.round_to_mintick(_y)), style = label.style_label_right, color = color.new(#0d5b66, 5), textcolor = #aef3fb, size = size.small)

    // ── Conversation panel ──────────────────────────────────────────────────
    if showPanel
        if na(netPanel)
            netPanel := table.new(position.top_right, 2, 22, bgcolor = color.new(#0b0b1c, 6), frame_color = color.new(#7b2fff, 30), frame_width = 2, border_width = 0)
        table.cell(netPanel, 0, 0, "INVISIBLE NETWORK", text_color = #b388ff, text_size = size.normal)
        table.cell(netPanel, 1, 0, netBias == 1 ? "▲ BULL" : netBias == -1 ? "▼ BEAR" : "○ FLAT", text_color = netBias == 1 ? #5aff9e : netBias == -1 ? #ff5a5a : color.gray, text_size = size.normal)
        table.cell(netPanel, 0, 1, "Path confidence", text_color = color.silver, text_size = size.small)
        table.cell(netPanel, 1, 1, str.tostring(math.round(_routeConf)) + "%", text_color = _routeConf >= 60 ? #5aff9e : color.yellow, text_size = size.small)
        table.cell(netPanel, 0, 2, "Memory", text_color = color.silver, text_size = size.small)
        table.cell(netPanel, 1, 2, str.tostring(_openMem) + " open · " + str.tostring(_consumedCnt) + " consumed", text_color = color.white, text_size = size.small)
        // magnet-sorted list of open eligible nodes (used for both sections)
        array<int> _mg = array.new_int()
        if array.size(nPx) > 0
            for _i = 0 to array.size(nPx) - 1
                if array.get(nState, _i) <= 1 and f_auth(_i) >= authMin
                    array.push(_mg, _i)
        if array.size(_mg) > 1
            for _a = 1 to array.size(_mg) - 1
                int   _key = array.get(_mg, _a)
                float _km  = f_magnet(_key)
                int   _b   = _a - 1
                bool  _go  = _b >= 0
                while _go
                    if f_magnet(array.get(_mg, _b)) < _km
                        array.set(_mg, _b + 1, array.get(_mg, _b))
                        _b -= 1
                        _go := _b >= 0
                    else
                        _go := false
                array.set(_mg, _b + 1, _key)
        // STRONGEST MAGNETS (top 3 by magnet score)
        table.cell(netPanel, 0, 3, "─ STRONGEST MAGNETS ─", text_color = #ff7fe6, text_size = size.small)
        table.cell(netPanel, 1, 3, "", text_color = color.white, text_size = size.small)
        int _mn = math.min(array.size(_mg), 3)
        for _k = 0 to 2
            if _k < _mn
                int _idx = array.get(_mg, _k)
                int _nd  = array.get(nDir, _idx)
                table.cell(netPanel, 0, 4 + _k, str.tostring(_k + 1) + " " + f_wtTf(array.get(nWt, _idx)) + (_nd == 1 ? " ▲ " : " ▼ ") + str.tostring(math.round_to_mintick(array.get(nPx, _idx))), text_color = _nd == 1 ? #5aff9e : #ff5a5a, text_size = size.small)
                table.cell(netPanel, 1, 4 + _k, "MAGNET " + str.tostring(math.round(f_magnet(_idx))), text_color = #ffd000, text_size = size.small)
            else
                table.cell(netPanel, 0, 4 + _k, "—", text_color = color.gray, text_size = size.small)
                table.cell(netPanel, 1, 4 + _k, "", text_color = color.gray, text_size = size.small)
        // ACTIVE CONVERSATIONS (top 5, with status + distance)
        table.cell(netPanel, 0, 7, "─ ACTIVE CONVERSATIONS ─", text_color = #ff7fe6, text_size = size.small)
        table.cell(netPanel, 1, 7, "", text_color = color.white, text_size = size.small)
        int _cn = math.min(array.size(_mg), 5)
        for _k = 0 to 4
            if _k < _cn
                int   _idx  = array.get(_mg, _k)
                int   _nd   = array.get(nDir, _idx)
                float _np   = array.get(nPx, _idx)
                float _dATR = math.abs(close - _np) / math.max(_atr, 1e-10)
                int   _rv   = array.get(nRev, _idx)
                bool  _ahead = netBias == 1 ? _np > close : _np < close
                string _status = _dATR < 0.5 ? "ATTACKING" : _rv > 0 ? "RETURNING" : _ahead ? "ATTRACTING" : "BEHIND"
                table.cell(netPanel, 0, 8 + _k, f_wtTf(array.get(nWt, _idx)) + (_nd == 1 ? " ▲ " : " ▼ ") + str.tostring(math.round_to_mintick(_np)), text_color = _nd == 1 ? #5aff9e : #ff5a5a, text_size = size.small)
                table.cell(netPanel, 1, 8 + _k, _status + "  " + str.tostring(_dATR, "#.#") + " ATR", text_color = color.white, text_size = size.small)
            else
                table.cell(netPanel, 0, 8 + _k, "—", text_color = color.gray, text_size = size.small)
                table.cell(netPanel, 1, 8 + _k, "", text_color = color.gray, text_size = size.small)
        // DOMINANT PATH
        table.cell(netPanel, 0, 13, "─ DOMINANT PATH ─", text_color = #ff7fe6, text_size = size.small)
        table.cell(netPanel, 1, 13, str.tostring(math.round(_routeConf)) + "%", text_color = _routeConf >= 60 ? #5aff9e : color.yellow, text_size = size.small)
        table.cell(netPanel, 0, 14, _domPath, text_color = #b388ff, text_size = size.small)
        table.cell(netPanel, 1, 14, "", text_color = color.white, text_size = size.small)
